}); } function dismissReadingResume(){document.getElementById('readingResume')?.classList.remove('show');} window.addEventListener('scroll',()=>{if(readingPositionTimer) return; readingPositionTimer=setTimeout(()=>{readingPositionTimer=null;saveReadingPosition()},250)},{passive:true});
